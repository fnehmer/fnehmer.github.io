We will solve this challenge with python. We will analyse the letter frequency and pattern frequency in the found files.

`freq_analyser.py`
```python
import sys
from collections import Counter, OrderedDict

data = sys.stdin.read().replace(" ", "")

print("### 10 most frequent letters ###")
print(str(Counter(data).most_common(10)) + "\n")


print("### 10 most frequent bi-patterns ###")
bi_pattern_list = []
for i in range(len(data) - 2):
    bi_pattern_list.append(data[i] + data[i+1])

print(str(Counter(bi_pattern_list).most_common(10)) + "\n")


print("### 10 most frequent tri-patterns ###")
tri_pattern_list = []
for i in range(len(data) - 3):
    tri_pattern_list.append(data[i] + data[i+1] + data[i+2])

print(str(Counter(tri_pattern_list).most_common(10)) + "\n")

print("### 10 most frequent quad-patterns ###")
quad_pattern_list = []
for i in range(len(data) - 4):
    quad_pattern_list.append(data[i] + data[i+1] + data[i+2] + data[i+3])

print(str(Counter(quad_pattern_list).most_common(10)) + "\n")

print("### most frequent doubles ###")
do_pattern_list = []
for i in range(len(data) - 2):
    if(data[i] == data[i+1]):
        do_pattern_list.append(data[i] + data[i+1])

print(str(Counter(do_pattern_list).most_common()) + "\n")
```
